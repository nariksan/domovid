// Загрузка счётчика непрочитанных сообщений в мессенджере let messengerUnsubscribe = null; function loadMessengerBadge() { if (!currentUser) return; if (messengerUnsubscribe) messengerUnsubscribe(); messengerUnsubscribe = db.collection('chats') .where('participants', 'array-contains', currentUser.id) .onSnapshot((snapshot) => { let totalUnread = 0; snapshot.docs.forEach(doc => { const chat = doc.data(); const unread = (chat.unread && chat.unread[currentUser.id]) || 0; totalUnread += unread; }); const badge = document.getElementById('messengerBadge'); if (badge) { if (totalUnread > 0) { badge.textContent = totalUnread > 99 ? '99+' : totalUnread; badge.style.display = 'block'; } else { badge.style.display = 'none'; } } }, (error) => { console.error('Ошибка загрузки badge:', error); }); }
